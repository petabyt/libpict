287inline static int ptp_write_u8(void *buf, uint8_t out) {
288 ((uint8_t *)buf)[0] = out;
289 return 1;
290}
291inline static int ptp_write_u16(void *buf, uint16_t out) {
292 uint8_t *b = (uint8_t *)buf;
293 b[0] = out & 0xFF;
294 b[1] = (out >> 8) & 0xFF;
295 return 2;
296}
297inline static int ptp_write_u32(void *buf, uint32_t out) {
298 uint8_t *b = (uint8_t *)buf;
299 b[0] = out & 0xFF;
300 b[1] = (out >> 8) & 0xFF;
301 b[2] = (out >> 16) & 0xFF;
302 b[3] = (out >> 24) & 0xFF;
303 return 4;
304}
305inline static int ptp_read_u8(const void *buf, uint8_t *out) {
306 const uint8_t *b = (const uint8_t *)buf;
307 *out = b[0];
308 return 1;
309}
310inline static int ptp_read_u16(const void *buf, uint16_t *out) {
311 const uint8_t *b = (const uint8_t *)buf;
312 *out = (uint16_t)b[0] | ((uint16_t)b[1] << 8);
313 return 2;
314}
315inline static int ptp_read_u32(const void *buf, uint32_t *out) {
316 const uint8_t *b = (const uint8_t *)buf;
317 *out = (uint32_t)b[0] | ((uint32_t)b[1] << 8) | ((uint32_t)b[2] << 16) | ((uint32_t)b[3] << 24);
318 return 4;
319}
320inline static int ptp_read_u64(const void *buf, uint64_t *out) {
321 uint32_t lo, hi;
322 ptp_read_u32(buf, &lo);
323 ptp_read_u32((const uint8_t *)buf + 4, &hi);
324 *out = ((uint64_t)hi << 32) | lo;
325 return 8;
326}
